% \djsLastUpdated{<date>}, called once after the setup line, sets the
% colophon at the foot of the first page only: the title block's accent rule
% (the same \djs@accentrule, so the two always match), then the date on the
% left and a link to the site on the right, all inside the bottom margin.
% Every later page carries its number over the page count at the bottom right,
% under the header's link. The count is the kernel's \PreviousTotalPages, which
% every run already records in the .aux file. Sizes are fixed, so these
% footers match the Further Maths packs'. Both are drawn over the page rather
% than set as footers, so no page style changes.
\makeatletter
\newcommand{\djsLastUpdated}[1]{%
  \AtBeginDocument{\AddToHookNext{shipout/foreground}{\djs@colophon{#1}}}}
\newcommand{\djs@footR}[1]{%
  \put(\dimexpr\paperwidth-1in\relax,-\dimexpr\paperheight-0.32in\relax){%
    \makebox[0pt][r]{#1}}}
\newcommand{\djs@colophon}[1]{%
  \put(0.8in,-\dimexpr\paperheight-0.47in\relax){%
    \makebox[0pt][l]{\djs@accentrule{\textwidth}}}%
  \put(0.8in,-\dimexpr\paperheight-0.32in\relax){\makebox[0pt][l]{%
    \fontsize{9}{11}\selectfont Last updated #1}}%
  \djs@footR{\fontsize{7}{8.5}\selectfont\color{djsMuted}%
    Latest version at \href{https://resources.danieljsmith.org/}%
      {\color{black}resources.danieljsmith.org}}}
\AddToHook{shipout/foreground}{%
  \ifnum\value{page}>1
    \djs@footR{{\fontsize{9}{11}\selectfont\thepage}%
      {\fontsize{7}{8.5}\selectfont\color{djsMuted}%
        \hspace{0.3em}/\hspace{0.3em}\PreviousTotalPages}}%
  \fi}
\makeatother

% The solution body: a white breakable box with a left scarlet rule. The
% rule continues onto a continuation page and stops where the solution
% stops, then turns into a short horizontal of the same weight so the
% solution visibly ends. The argument is the answer label.
\newcommand{\djsSolutionAnswer}[1]{%
  {\color{scarlet}\bfseries Answer\ \ #1}}
\newtcolorbox{djsSolution}[1]{%
  enhanced, breakable, frame hidden, sharp corners, colback=white,
  height fixed for=first and middle, height fill,
  extras unbroken and last={natural height},
  borderline west={2.2pt}{0pt}{scarlet},
  left=11pt, right=0pt, top=5pt, bottom=13pt,
  before skip=13pt, after skip=4pt,
  overlay unbroken and last={%
    \fill[scarlet] (frame.south west)
      rectangle ([xshift=3.5cm, yshift=2.2pt]frame.south west);},
  before upper={\setlength{\parskip}{6pt}%
                {\color{scarlet}\bfseries Solution}%
                \hfill\djsSolutionAnswer{#1}\par\vspace{4pt}}}
