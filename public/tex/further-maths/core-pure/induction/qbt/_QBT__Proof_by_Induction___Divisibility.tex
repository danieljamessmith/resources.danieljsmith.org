% ---- Question 4 ----
\questionitem
% [Source: _QBT__Proof_by_Induction___Divisibility, Q4]

\[
2^n+2\cdot 3^{2n}+4^{2n+1}
\]
is divisible by $7$ \marks{6}

\newpage

% ---- Question 5 ----

\[u_n=5^{2n}-2^n\]

\newpage
