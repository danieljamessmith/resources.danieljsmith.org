% ---- Question 1 ----
\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q1]

Prove by induction that, for $n \in \mathbb{Z}^+$,
\[
\renewcommand{\arraystretch}{1.2}
\begin{pmatrix} 2 & 1 \\ 0 & 3 \end{pmatrix}^n = \begin{pmatrix} 2^n & 3^n - 2^n \\ 0 & 3^n \end{pmatrix}
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
Let
\[
\mathbf{A}=\begin{pmatrix}2&1\\0&3\end{pmatrix}
\]

We prove by induction that
\[
\mathbf{A}^n=\begin{pmatrix}2^n&3^n-2^n\\0&3^n\end{pmatrix}\quad\text{for all }n\in\mathbb{Z}^+
\]

\textbf{Base case:} $n=1$. The left-hand side is
\[
\mathbf{A}^1=\mathbf{A}=\begin{pmatrix}2&1\\0&3\end{pmatrix}
\]

The right-hand side when $n=1$ is
\[
\begin{pmatrix}2^1&3^1-2^1\\0&3^1\end{pmatrix}
=
\begin{pmatrix}2&1\\0&3\end{pmatrix}
\]

So the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k\in\mathbb{Z}^+$. That is, assume
\[
\mathbf{A}^k=\begin{pmatrix}2^k&3^k-2^k\\0&3^k\end{pmatrix}
\]

\textbf{Inductive step:} We must show that this implies
\[
\mathbf{A}^{k+1}=\begin{pmatrix}2^{k+1}&3^{k+1}-2^{k+1}\\0&3^{k+1}\end{pmatrix}
\]

Using $\mathbf{A}^{k+1}=\mathbf{A}^k\mathbf{A}$ and the inductive hypothesis:
\begin{align*}
\mathbf{A}^{k+1}
&=
\begin{pmatrix}2^k&3^k-2^k\\0&3^k\end{pmatrix}
\begin{pmatrix}2&1\\0&3\end{pmatrix} \\
&=
\begin{pmatrix}
2^k\times2+(3^k-2^k)\times0 & 2^k\times1+(3^k-2^k)\times3 \\
0\times2+3^k\times0 & 0\times1+3^k\times3
\end{pmatrix} \\
&=
\begin{pmatrix}
2^{k+1} & 2^k+3(3^k-2^k) \\
0 & 3^{k+1}
\end{pmatrix}
\end{align*}

Now simplify the top-right entry:
\begin{align*}
2^k+3(3^k-2^k)
&=2^k+3^{k+1}-3\times2^k \\
&=3^{k+1}-2\times2^k \\
&=3^{k+1}-2^{k+1}
\end{align*}

So
\[
\mathbf{A}^{k+1}=
\begin{pmatrix}
2^{k+1} & 3^{k+1}-2^{k+1} \\
0 & 3^{k+1}
\end{pmatrix}
\]

This is exactly the required form, so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all $n\in\mathbb{Z}^+$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q2]

Prove by mathematical induction that, for $n \in \mathbb{N}$,
\[
\begin{pmatrix}
3 & -1 \\
1 & 1
\end{pmatrix}^n = 2^{n-1}
\begin{pmatrix}
n+2 & -n \\
n & 2-n
\end{pmatrix}
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
Let
\[
\mathbf{A}=\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}
\]

We will prove by mathematical induction that
\[
\mathbf{A}^n=2^{n-1}\begin{pmatrix}
n+2 & -n\\
n & 2-n
\end{pmatrix}
\qquad \text{for all } n\in\mathbb{N}
\]

\textbf{Base case:} \(n=1\). The left-hand side is
\[
\mathbf{A}^1=\mathbf{A}=\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}
\]

Now evaluate the right-hand side when \(n=1\):
\[
2^{1-1}\begin{pmatrix}
1+2 & -1\\
1 & 2-1
\end{pmatrix}
=
1\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}
=
\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}
\]

So the statement is true for \(n=1\).

\textbf{Inductive hypothesis:} Assume the statement is true for \(n=k\), where \(k\in\mathbb{N}\). That is, assume
\[
\mathbf{A}^k=2^{k-1}\begin{pmatrix}
k+2 & -k\\
k & 2-k
\end{pmatrix}
\]

\textbf{Inductive step:} We must show that this implies
\[
\mathbf{A}^{k+1}=2^k\begin{pmatrix}
(k+1)+2 & -(k+1)\\
k+1 & 2-(k+1)
\end{pmatrix}
\]

Using \(\mathbf{A}^{k+1}=\mathbf{A}^k\mathbf{A}\) and the inductive hypothesis:
\begin{align*}
\mathbf{A}^{k+1}
&=\mathbf{A}^k\mathbf{A}\\
&=\left(2^{k-1}\begin{pmatrix}
k+2 & -k\\
k & 2-k
\end{pmatrix}\right)
\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}\\
&=2^{k-1}
\begin{pmatrix}
k+2 & -k\\
k & 2-k
\end{pmatrix}
\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}
\end{align*}

Now multiply the matrices:
\begin{align*}
\begin{pmatrix}
k+2 & -k\\
k & 2-k
\end{pmatrix}
\begin{pmatrix}
3 & -1\\
1 & 1
\end{pmatrix}
&=
\begin{pmatrix}
3(k+2)+(-k)(1) & (k+2)(-1)+(-k)(1)\\
3k+(2-k)(1) & k(-1)+(2-k)(1)
\end{pmatrix}\\
&=
\begin{pmatrix}
3k+6-k & -k-2-k\\
3k+2-k & -k+2-k
\end{pmatrix}\\
&=
\begin{pmatrix}
2k+6 & -2k-2\\
2k+2 & 2-2k
\end{pmatrix}
\end{align*}

So
\[
\mathbf{A}^{k+1}
=
2^{k-1}
\begin{pmatrix}
2k+6 & -2k-2\\
2k+2 & 2-2k
\end{pmatrix}
\]

Factor out \(2\) from the matrix:
\[
\mathbf{A}^{k+1}
=
2^{k}
\begin{pmatrix}
k+3 & -(k+1)\\
k+1 & 1-k
\end{pmatrix}
\]

Since \(1-k=2-(k+1)\), this becomes
\[
\mathbf{A}^{k+1}
=
2^{k}
\begin{pmatrix}
k+3 & -(k+1)\\
k+1 & 2-(k+1)
\end{pmatrix}
\]

This is exactly the required form, so the statement is true for \(n=k+1\).

\textbf{Conclusion:} If the statement is true for \(n=k\), then it is true for \(n=k+1\). Since it is true for \(n=1\), by the principle of mathematical induction the statement is true for all \(n\in\mathbb{N}\).
\end{djsSolution}

% ---- Question 3 ----
\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q3]

The matrix $\mathbf{A}$ is given by
\[
\mathbf{A} = \begin{pmatrix} 4 & 2 \\ -2 & 0 \end{pmatrix}
\]
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Prove by induction that
  \[
  \mathbf{A}^n = 2^n \begin{pmatrix} n+1 & n \\ -n & 1-n \end{pmatrix}
  \]
  for all positive integers $n$. \marks{6} \\
  \item The matrix $\mathbf{B}$ is given by
  \[
  \mathbf{B} = (\mathbf{A}^n)^{-1}
  \]
  Hence find $\mathbf{B}$ in terms of $n$. \marks{2} \\
\end{enumerate}

\vspace*{10pt}

\begin{djsSolution}
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item We prove by induction that
\[
\mathbf{A}^n=2^n\begin{pmatrix}n+1&n\\-n&1-n\end{pmatrix}
\]

for all positive integers $n$.

\textbf{Base case:} $n=1$. The claimed formula gives
\[
\mathbf{A}^1=2^1\begin{pmatrix}2&1\\-1&0\end{pmatrix}
=\begin{pmatrix}4&2\\-2&0\end{pmatrix}
=\mathbf{A}
\]

So the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k$ is a positive integer. That is, assume
\[
\mathbf{A}^k=2^k\begin{pmatrix}k+1&k\\-k&1-k\end{pmatrix}
\]

\textbf{Inductive step:} We must show that
\[
\mathbf{A}^{k+1}=2^{k+1}\begin{pmatrix}k+2&k+1\\-(k+1)&1-(k+1)\end{pmatrix}
\]

Using $\mathbf{A}^{k+1}=\mathbf{A}^k\mathbf{A}$ and the inductive hypothesis:
\begin{align*}
\mathbf{A}^{k+1} &= \mathbf{A}^k\mathbf{A} \\
&= 2^k\begin{pmatrix}k+1&k\\-k&1-k\end{pmatrix}
\begin{pmatrix}4&2\\-2&0\end{pmatrix}
\end{align*}

Now multiply the matrices:
\begin{align*}
\mathbf{A}^{k+1}
&=2^k\begin{pmatrix}
4(k+1)-2k & 2(k+1)+0 \\
-4k-2(1-k) & -2k+0
\end{pmatrix} \\
&=2^k\begin{pmatrix}
2k+4 & 2k+2 \\
-2k-2 & -2k
\end{pmatrix}
\end{align*}

Factor out $2$ from the matrix:
\begin{align*}
\mathbf{A}^{k+1}
&=2^{k+1}\begin{pmatrix}
k+2 & k+1 \\
-(k+1) & -k
\end{pmatrix}
\end{align*}

Since
\[
-k = 1-(k+1)
\]

this becomes
\[
\mathbf{A}^{k+1}=2^{k+1}\begin{pmatrix}k+2&k+1\\-(k+1)&1-(k+1)\end{pmatrix}
\]

This is exactly the required form, so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all positive integers $n$.

  \item Let
\[
\mathbf{M}=\begin{pmatrix}n+1&n\\-n&1-n\end{pmatrix}
\]

so that part (a) gives
\[
\mathbf{A}^n=2^n\mathbf{M}
\]

Hence
\[
\mathbf{B}=(\mathbf{A}^n)^{-1}=(2^n\mathbf{M})^{-1}=2^{-n}\mathbf{M}^{-1}
\]

Now find $\mathbf{M}^{-1}$. The inverse of a $2\times2$ matrix is
\[
\begin{pmatrix}a&b\\c&d\end{pmatrix}^{-1}
=\frac{1}{ad-bc}\begin{pmatrix}d&-b\\-c&a\end{pmatrix}
\]

First calculate the determinant:
\begin{align*}
\det(\mathbf{M})
&=(n+1)(1-n)-n(-n) \\
&=(n+1)(1-n)+n^2 \\
&=1-n^2+n^2 \\
&=1
\end{align*}

So
\[
\mathbf{M}^{-1}=\begin{pmatrix}1-n&-n\\n&n+1\end{pmatrix}
\]

Therefore
\[
\mathbf{B}=\frac{1}{2^n}\begin{pmatrix}1-n&-n\\n&n+1\end{pmatrix}
\]

  \end{enumerate}
\end{djsSolution}

% ---- Question 4 ----
\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q4]

The matrix $\mathbf{A}$ is given by
\[
\mathbf{A} = \begin{pmatrix} 2 & 0 & 0 \\ 1 & 2 & 0 \\ 0 & 1 & 2 \end{pmatrix}
\]
Prove by induction that, for $n \geq 1$,
\[
\renewcommand{\arraystretch}{1.2}
\mathbf{A}^n = \begin{pmatrix} 2^n & 0 & 0 \\ n\,2^{n-1} & 2^n & 0 \\ \frac{n(n-1)}{2}\,2^{n-2} & n\,2^{n-1} & 2^n \end{pmatrix}
\]
\marks[-30pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We prove by induction that
\[
\mathbf{A}^n=
\begin{pmatrix}
2^n & 0 & 0\\
n\,2^{n-1} & 2^n & 0\\
\frac{n(n-1)}{2}\,2^{n-2} & n\,2^{n-1} & 2^n
\end{pmatrix}
\]

\textbf{Base case:} \(n=1\). The right-hand side is
\[
\begin{pmatrix}
2^1 & 0 & 0\\
1\times 2^0 & 2^1 & 0\\
\frac{1(1-1)}{2}\,2^{-1} & 1\times 2^0 & 2^1
\end{pmatrix}
=
\begin{pmatrix}
2 & 0 & 0\\
1 & 2 & 0\\
0 & 1 & 2
\end{pmatrix}
= \mathbf{A}
\]

So the statement is true for \(n=1\).

\textbf{Inductive hypothesis:} Assume the statement is true for \(n=k\), where \(k\) is an integer with \(k\geq 1\). That is, assume
\[
\mathbf{A}^k=
\begin{pmatrix}
2^k & 0 & 0\\
k\,2^{k-1} & 2^k & 0\\
\frac{k(k-1)}{2}\,2^{k-2} & k\,2^{k-1} & 2^k
\end{pmatrix}
\]

\textbf{Inductive step:} We must show that the statement is true for \(n=k+1\).

Using \(\mathbf{A}^{k+1}=\mathbf{A}^k\mathbf{A}\) and the inductive hypothesis:
\begin{align*}
\mathbf{A}^{k+1}
&=
\begin{pmatrix}
2^k & 0 & 0\\
k\,2^{k-1} & 2^k & 0\\
\frac{k(k-1)}{2}\,2^{k-2} & k\,2^{k-1} & 2^k
\end{pmatrix}
\begin{pmatrix}
2 & 0 & 0\\
1 & 2 & 0\\
0 & 1 & 2
\end{pmatrix} \\
&=
\begin{pmatrix}
2^k\times 2 & 0 & 0\\[1ex]
k\,2^{k-1}\times 2 + 2^k\times 1 & 2^k\times 2 & 0\\[1ex]
\frac{k(k-1)}{2}\,2^{k-2}\times 2 + k\,2^{k-1}\times 1 &
k\,2^{k-1}\times 2 + 2^k\times 1 &
2^k\times 2
\end{pmatrix} \\
&=
\begin{pmatrix}
2^{k+1} & 0 & 0\\
k\,2^k + 2^k & 2^{k+1} & 0\\
\frac{k(k-1)}{2}\,2^{k-1} + k\,2^{k-1} & k\,2^k + 2^k & 2^{k+1}
\end{pmatrix} \\
&=
\begin{pmatrix}
2^{k+1} & 0 & 0\\
(k+1)2^k & 2^{k+1} & 0\\
\left(\frac{k(k-1)}{2}+k\right)2^{k-1} & (k+1)2^k & 2^{k+1}
\end{pmatrix}
\end{align*}

Now simplify the bottom-left entry:
\begin{align*}
\left(\frac{k(k-1)}{2}+k\right)2^{k-1}
&=
\left(\frac{k(k-1)+2k}{2}\right)2^{k-1} \\
&=
\left(\frac{k^2-k+2k}{2}\right)2^{k-1} \\
&=
\frac{k(k+1)}{2}\,2^{k-1}
\end{align*}

So
\[
\mathbf{A}^{k+1}
=
\begin{pmatrix}
2^{k+1} & 0 & 0\\
(k+1)2^k & 2^{k+1} & 0\\
\frac{k(k+1)}{2}\,2^{k-1} & (k+1)2^k & 2^{k+1}
\end{pmatrix}
\]

This is exactly the required form with \(n=k+1\), since
\[
(k+1)2^k=(k+1)2^{(k+1)-1}
\]

and
\[
\frac{k(k+1)}{2}\,2^{k-1}
=
\frac{(k+1)\bigl((k+1)-1\bigr)}{2}\,2^{(k+1)-2}
\]

so the statement is true for \(n=k+1\).

\textbf{Conclusion:} If the statement is true for \(n=k\), then it is true for \(n=k+1\). Since it is true for \(n=1\), by the principle of mathematical induction the statement is true for all \(n\geq 1\).
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q5]

The matrix $\mathbf{M}$ is given by $\mathbf{M} = \begin{pmatrix}3c & 1 \\ 0 & 3c\end{pmatrix}$, where $c$ is a positive constant. \\
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Prove by induction that for all positive integers $n$,
  \[
  \mathbf{M}^n = \begin{pmatrix}(3c)^n & n(3c)^{n-1} \\ 0 & (3c)^n\end{pmatrix}
  \]
  \marks[-20pt]{6}
  \item Given that $\det(\mathbf{M}^n) = 144^n$, find the value of $c$. \marks{5} \\
\end{enumerate}

\vspace*{10pt}

\begin{djsSolution}
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item We prove the result by induction on $n$.

  \textbf{Base case:} $n=1$. The left-hand side is
  \[
  \mathbf{M}^1=\mathbf{M}=\begin{pmatrix}3c&1\\0&3c\end{pmatrix}
  \]

  The right-hand side when $n=1$ is
  \[
  \begin{pmatrix}(3c)^1 & 1(3c)^{1-1}\\0&(3c)^1\end{pmatrix}
  =
  \begin{pmatrix}(3c)^1 & 1(3c)^0\\0&(3c)^1\end{pmatrix}
  =
  \begin{pmatrix}3c&1\\0&3c\end{pmatrix}
  \]

  So the statement is true for $n=1$.

  \textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k$ is a positive integer. That is, assume
  \[
  \mathbf{M}^k=\begin{pmatrix}(3c)^k & k(3c)^{k-1}\\0&(3c)^k\end{pmatrix}
  \]

  \textbf{Inductive step:} We must show that this implies
  \[
  \mathbf{M}^{k+1}=\begin{pmatrix}(3c)^{k+1} & (k+1)(3c)^k\\0&(3c)^{k+1}\end{pmatrix}
  \]

  Using $\mathbf{M}^{k+1}=\mathbf{M}^k\mathbf{M}$ and the inductive hypothesis:
  \begin{align*}
  \mathbf{M}^{k+1}
  &= \mathbf{M}^k\mathbf{M} \\
  &= \begin{pmatrix}(3c)^k & k(3c)^{k-1}\\0&(3c)^k\end{pmatrix}
     \begin{pmatrix}3c&1\\0&3c\end{pmatrix} \\
  &= \begin{pmatrix}
  (3c)^k\times 3c + k(3c)^{k-1}\times 0
  &
  (3c)^k\times 1 + k(3c)^{k-1}\times 3c
  \\
  0\times 3c + (3c)^k\times 0
  &
  0\times 1 + (3c)^k\times 3c
  \end{pmatrix} \\
  &= \begin{pmatrix}
  (3c)^{k+1}
  &
  (3c)^k + 3ck(3c)^{k-1}
  \\
  0
  &
  (3c)^{k+1}
  \end{pmatrix}
  \end{align*}

  Now simplify the top-right entry:
  \begin{align*}
  (3c)^k + 3ck(3c)^{k-1}
  &= (3c)^k + k(3c)^k \\
  &= (k+1)(3c)^k
  \end{align*}

  Hence
  \[
  \mathbf{M}^{k+1}
  =
  \begin{pmatrix}(3c)^{k+1} & (k+1)(3c)^k\\0&(3c)^{k+1}\end{pmatrix}
  \]

  This is exactly the required form, so the statement is true for $n=k+1$.

  \textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all positive integers $n$.

  \item Part (a) gives
  \[
  \mathbf{M}^n=\begin{pmatrix}(3c)^n & n(3c)^{n-1}\\0&(3c)^n\end{pmatrix}
  \]

  Its determinant is the product of the leading diagonal entries minus the product of the other two, so
  \begin{align*}
  \det(\mathbf{M}^n)
  &= (3c)^n\times(3c)^n-n(3c)^{n-1}\times0 \\
  &= (3c)^{2n} \\
  &= (9c^2)^n
  \end{align*}

  We are given that
  \[
  \det(\mathbf{M}^n)=144^n
  \]

  So
  \[
  (9c^2)^n=144^n
  \]

  Hence
  \[
  9c^2=144
  \]

  Therefore
  \[
  c^2=16
  \]

  Since $c$ is positive, taking the positive root gives
  \[
  c=4
  \]

\end{enumerate}
\end{djsSolution}

% ---- Question 6 ----
\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q6]

For $n \in \mathbb{Z}^{+}$, show, using mathematical induction, that
\[
\renewcommand{\arraystretch}{1.4}
\begin{pmatrix}

-1 & 1 & 0 \\
0 & -1 & 1 \\
0 & 0 & -1
\end{pmatrix}^{n}
=
\begin{pmatrix}
(-1)^{n} & -n(-1)^{n} & \frac{n(n-1)}{2}(-1)^{n} \\
0 & (-1)^{n} & -n(-1)^{n} \\
0 & 0 & (-1)^{n}
\end{pmatrix}
\]
\marks[-30pt]{6}

\vspace*{10pt}

\begin{djsSolution}
Let
\[
\mathbf{A}=
\begin{pmatrix}
-1 & 1 & 0\\
0 & -1 & 1\\
0 & 0 & -1
\end{pmatrix}
\]

We prove by induction that
\[
\mathbf{A}^{n}=
\begin{pmatrix}
(-1)^{n} & -n(-1)^{n} & \frac{n(n-1)}{2}(-1)^{n}\\
0 & (-1)^{n} & -n(-1)^{n}\\
0 & 0 & (-1)^{n}
\end{pmatrix}
\]

\textbf{Base case:} \(n=1\). Substituting \(n=1\) into the right-hand side gives
\[
\begin{pmatrix}
(-1)^{1} & -1(-1)^{1} & \frac{1(1-1)}{2}(-1)^{1}\\
0 & (-1)^{1} & -1(-1)^{1}\\
0 & 0 & (-1)^{1}
\end{pmatrix}
=
\begin{pmatrix}
-1 & 1 & 0\\
0 & -1 & 1\\
0 & 0 & -1
\end{pmatrix}
=\mathbf{A}
\]

So the statement is true for \(n=1\).

\textbf{Inductive hypothesis:} Assume the statement is true for \(n=k\), where \(k\in\mathbb{Z}^+\). That is, assume
\[
\mathbf{A}^{k}=
\begin{pmatrix}
(-1)^{k} & -k(-1)^{k} & \frac{k(k-1)}{2}(-1)^{k}\\
0 & (-1)^{k} & -k(-1)^{k}\\
0 & 0 & (-1)^{k}
\end{pmatrix}
\]

\textbf{Inductive step:} We must show that the statement is true for \(n=k+1\). For simplicity, let
\[
a=(-1)^{k}, \qquad b=-k(-1)^{k}, \qquad c=\dfrac{k(k-1)}{2}(-1)^{k}
\]

Then
\[
\mathbf{A}^{k}=
\begin{pmatrix}
a & b & c\\
0 & a & b\\
0 & 0 & a
\end{pmatrix}
\]

Now multiply by \(\mathbf{A}\):
\begin{align*}
\mathbf{A}^{k+1}=\mathbf{A}^{k}\mathbf{A}
&=
\begin{pmatrix}
a & b & c\\
0 & a & b\\
0 & 0 & a
\end{pmatrix}
\begin{pmatrix}
-1 & 1 & 0\\
0 & -1 & 1\\
0 & 0 & -1
\end{pmatrix}\\
&=
\begin{pmatrix}
-a & a-b & b-c\\
0 & -a & a-b\\
0 & 0 & -a
\end{pmatrix}
\end{align*}

Now simplify each entry.

First, the entries on the leading diagonal are
\[
-a=-(-1)^{k}=(-1)^{k+1}
\]

Next, the entries just above the leading diagonal are
\begin{align*}
a-b
&= (-1)^{k}-\bigl(-k(-1)^{k}\bigr)\\
&= (-1)^{k}+k(-1)^{k}\\
&= (k+1)(-1)^{k}\\
&= -(k+1)(-1)^{k+1}
\end{align*}

Finally, the top-right entry is
\begin{align*}
b-c
&= -k(-1)^{k}-\dfrac{k(k-1)}{2}(-1)^{k}\\
&= \left(-k-\dfrac{k(k-1)}{2}\right)(-1)^{k}\\
&= -\dfrac{2k+k(k-1)}{2}(-1)^{k}\\
&= -\dfrac{k^{2}+k}{2}(-1)^{k}\\
&= -\dfrac{k(k+1)}{2}(-1)^{k}\\
&= \dfrac{k(k+1)}{2}(-1)^{k+1}
\end{align*}

So
\[
\mathbf{A}^{k+1}
=
\begin{pmatrix}
(-1)^{k+1} & -(k+1)(-1)^{k+1} & \frac{k(k+1)}{2}(-1)^{k+1}\\
0 & (-1)^{k+1} & -(k+1)(-1)^{k+1}\\
0 & 0 & (-1)^{k+1}
\end{pmatrix}
\]

Since
\[
\dfrac{k(k+1)}{2}=\dfrac{(k+1)\bigl((k+1)-1\bigr)}{2}
\]

this is exactly the required form, so the statement is true for \(n=k+1\).

\textbf{Conclusion:} If the statement is true for \(n=k\), then it is true for \(n=k+1\). Since it is true for \(n=1\), by the principle of mathematical induction the statement is true for all \(n\in\mathbb{Z}^+\).
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q7]

The Fibonacci sequence $(F_n)$ is defined by $F_0=0$, $F_1=1$ and $F_{n+1}=F_n+F_{n-1}$ for $n\geq 1$. \\
You are given that
\[
\mathbf{M} = \begin{pmatrix} 1 & 1 \\ 1 & 0 \end{pmatrix}
\]
Prove by induction that
\[
\mathbf{M}^n = \begin{pmatrix} F_{n+1} & F_n \\ F_n & F_{n-1} \end{pmatrix}
\]
for all integers $n \geq 1$. \marks{6} \\

\vspace*{10pt}

\begin{djsSolution}
We prove by induction that
\[
\mathbf{M}^n=\begin{pmatrix}F_{n+1}&F_n\\[2pt]F_n&F_{n-1}\end{pmatrix}
\]

for all integers \(n\geq 1\).

\textbf{Base case:} \(n=1\). The left-hand side is
\[
\mathbf{M}^1=\mathbf{M}=\begin{pmatrix}1&1\\1&0\end{pmatrix}
\]

The recurrence gives
\[
F_2=F_1+F_0=1+0=1
\]

So
\[
\begin{pmatrix}1&1\\1&0\end{pmatrix}
=
\begin{pmatrix}F_2&F_1\\F_1&F_0\end{pmatrix}
\]

Hence
\[
\mathbf{M}=\begin{pmatrix}F_2&F_1\\F_1&F_0\end{pmatrix}
\]

So the statement is true for \(n=1\).

\textbf{Inductive hypothesis:} Assume the statement is true for \(n=k\), where \(k\) is an integer with \(k\geq 1\). That is, assume
\[
\mathbf{M}^k=\begin{pmatrix}F_{k+1}&F_k\\F_k&F_{k-1}\end{pmatrix}
\]

\textbf{Inductive step:} We must show that the statement is true for \(n=k+1\).

Using \(\mathbf{M}^{k+1}=\mathbf{M}^k\mathbf{M}\) and the inductive hypothesis:
\begin{align*}
\mathbf{M}^{k+1}
&=\mathbf{M}^k\mathbf{M}\\
&=\begin{pmatrix}F_{k+1}&F_k\\F_k&F_{k-1}\end{pmatrix}
\begin{pmatrix}1&1\\1&0\end{pmatrix}\\
&=\begin{pmatrix}
F_{k+1}\times 1+F_k\times 1 & F_{k+1}\times 1+F_k\times 0\\
F_k\times 1+F_{k-1}\times 1 & F_k\times 1+F_{k-1}\times 0
\end{pmatrix}\\
&=\begin{pmatrix}
F_{k+1}+F_k & F_{k+1}\\
F_k+F_{k-1} & F_k
\end{pmatrix}
\end{align*}

The Fibonacci recurrence gives
\[
F_{k+2}=F_{k+1}+F_k
\qquad\text{and}\qquad
F_{k+1}=F_k+F_{k-1}
\]

So
\[
\mathbf{M}^{k+1}
=\begin{pmatrix}
F_{k+2}&F_{k+1}\\
F_{k+1}&F_k
\end{pmatrix}
\]

This is exactly the required form, so the statement is true for \(n=k+1\).

\textbf{Conclusion:} If the statement is true for \(n=k\), then it is true for \(n=k+1\). Since it is true for \(n=1\), by the principle of mathematical induction the statement is true for all integers \(n\geq 1\).
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Matrices, Q8]

The matrix $\mathbf{A}$ is given by
\[
\mathbf{A} = \begin{pmatrix}
1 & 1 & 0 \\
0 & 2 & 1 \\
0 & 0 & 2
\end{pmatrix}
\]
Prove by induction that, for all positive integers $n$,
\[
\renewcommand{\arraystretch}{1.4}
\mathbf{A}^n = \begin{pmatrix}
1 & 2^n - 1 & 1 + (n-2)2^{n-1} \\
0 & 2^n & n\,2^{n-1} \\
0 & 0 & 2^n
\end{pmatrix}
\]
\marks[-30pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We prove the result by induction on $n$.

\textbf{Base case:} $n=1$. The claimed formula gives
\[
\mathbf{A}^1=
\begin{pmatrix}
1 & 2^1-1 & 1+(1-2)2^{1-1} \\
0 & 2^1 & 1\times 2^{1-1} \\
0 & 0 & 2^1
\end{pmatrix}
=
\begin{pmatrix}
1 & 1 & 0 \\
0 & 2 & 1 \\
0 & 0 & 2
\end{pmatrix}
=
\mathbf{A}
\]

So the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k$ is a positive integer. That is, assume
\[
\mathbf{A}^k=
\begin{pmatrix}
1 & 2^k-1 & 1+(k-2)2^{k-1} \\
0 & 2^k & k\,2^{k-1} \\
0 & 0 & 2^k
\end{pmatrix}
\]

\textbf{Inductive step:} We must show that the statement is true for $n=k+1$.

Using $\mathbf{A}^{k+1}=\mathbf{A}^k\mathbf{A}$ and the inductive hypothesis:
\begin{align*}
\mathbf{A}^{k+1}
&=
\begin{pmatrix}
1 & 2^k-1 & 1+(k-2)2^{k-1} \\
0 & 2^k & k\,2^{k-1} \\
0 & 0 & 2^k
\end{pmatrix}
\begin{pmatrix}
1 & 1 & 0 \\
0 & 2 & 1 \\
0 & 0 & 2
\end{pmatrix}
\end{align*}

Now multiply row by column:
\begin{align*}
\mathbf{A}^{k+1}
&=
\begin{pmatrix}
1 &
1+2(2^k-1) &
(2^k-1)+2\bigl(1+(k-2)2^{k-1}\bigr) \\
0 &
2^{k+1} &
2^k+2k\,2^{k-1} \\
0 &
0 &
2^{k+1}
\end{pmatrix}
\end{align*}

Now simplify each entry:
\begin{align*}
1+2(2^k-1) &= 1+2^{k+1}-2 = 2^{k+1}-1 \\
(2^k-1)+2\bigl(1+(k-2)2^{k-1}\bigr)
&= 2^k-1+2+(k-2)2^k \\
&= 1+(k-1)2^k \\
2^k+2k\,2^{k-1} &= 2^k+k\,2^k=(k+1)2^k
\end{align*}

So
\[
\mathbf{A}^{k+1}=
\begin{pmatrix}
1 & 2^{k+1}-1 & 1+(k-1)2^k \\
0 & 2^{k+1} & (k+1)2^k \\
0 & 0 & 2^{k+1}
\end{pmatrix}
\]

This is exactly the required form with $n=k+1$, since
\[
1+\bigl((k+1)-2\bigr)2^{(k+1)-1}=1+(k-1)2^k
\]

and
\[
(k+1)2^{(k+1)-1}=(k+1)2^k
\]

so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all positive integers $n$.
\end{djsSolution}
