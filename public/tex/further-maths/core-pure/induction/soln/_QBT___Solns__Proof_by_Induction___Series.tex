\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q1]

Prove by induction that, for $n \geq 1$
\[
\sum_{r=1}^{n} r\,2^r = 2 + (n-1)2^{n+1}
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We prove the result by mathematical induction.

\textbf{Base case:} $n=1$. The left-hand side is
\[
\sum_{r=1}^{1} r\,2^r = 1\times 2 = 2
\]

and the right-hand side is
\[
2+(1-1)2^{1+1}=2+0=2
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k \geq 1$. That is, assume
\[
\sum_{r=1}^{k} r\,2^r = 2 + (k-1)2^{k+1}
\]

\textbf{Inductive step:} Starting with the left-hand side for $n=k+1$:
\begin{align*}
\sum_{r=1}^{k+1} r\,2^r
&= \sum_{r=1}^{k} r\,2^r + (k+1)2^{k+1} \\
&= \left(2+(k-1)2^{k+1}\right) + (k+1)2^{k+1} \\
&= 2 + \bigl((k-1)+(k+1)\bigr)2^{k+1} \\
&= 2 + 2k\,2^{k+1} \\
&= 2 + k\,2^{k+2} \\
&= 2 + \bigl((k+1)-1\bigr)2^{(k+1)+1}
\end{align*}

This is the right-hand side with $n=k+1$, so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all $n \geq 1$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q2]

Prove by induction that, for $n \geq 1$
\[
\sum_{r=1}^{n} \left(2^r + 2 \times 3^r\right) = 2^{n+1} + 3^{n+1} - 5
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We use proof by induction on $n$.

\textbf{Base case:} $n=1$. The left-hand side is
\[
\sum_{r=1}^{1}\left(2^r+2\times 3^r\right)=2^1+2\times 3^1=2+6=8
\]

and the right-hand side is
\[
2^{1+1}+3^{1+1}-5=2^2+3^2-5=4+9-5=8
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k\geq 1$. That is, assume
\[
\sum_{r=1}^{k}\left(2^r+2\times 3^r\right)=2^{k+1}+3^{k+1}-5
\]

\textbf{Inductive step:} Starting with the left-hand side for $n=k+1$:
\[
\sum_{r=1}^{k+1}\left(2^r+2\times 3^r\right)
=\sum_{r=1}^{k}\left(2^r+2\times 3^r\right)+\left(2^{k+1}+2\times 3^{k+1}\right)
\]

Now use the inductive hypothesis:
\begin{align*}
\sum_{r=1}^{k+1}\left(2^r+2\times 3^r\right)
&=\left(2^{k+1}+3^{k+1}-5\right)+\left(2^{k+1}+2\times 3^{k+1}\right)\\
&=2^{k+1}+2^{k+1}+3^{k+1}+2\times 3^{k+1}-5\\
&=2^{k+2}+3\times 3^{k+1}-5\\
&=2^{k+2}+3^{k+2}-5
\end{align*}

This is the right-hand side with $n=k+1$, so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all $n \geq 1$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q3]

Prove by induction that for all positive integers $n$
\[
\sum_{r=1}^{n} r(r+1)(r+2)=\frac{1}{4}n(n+1)(n+2)(n+3)
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We prove the result by mathematical induction.

We want to show that for all positive integers $n$
\[
\sum_{r=1}^{n} r(r+1)(r+2)=\frac{1}{4}n(n+1)(n+2)(n+3)
\]

\textbf{Base case:} $n=1$. The left-hand side is
\[
\sum_{r=1}^{1} r(r+1)(r+2)=1(2)(3)=6
\]

and the right-hand side is
\[
\frac{1}{4}(1)(2)(3)(4)=6
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k$ is a positive integer. That is, assume
\[
\sum_{r=1}^{k} r(r+1)(r+2)=\frac{1}{4}k(k+1)(k+2)(k+3)
\]

\textbf{Inductive step:} Starting with the left-hand side for $n=k+1$:
\begin{align*}
\sum_{r=1}^{k+1} r(r+1)(r+2)
&=\sum_{r=1}^{k} r(r+1)(r+2) + (k+1)(k+2)(k+3) \\
&=\frac{1}{4}k(k+1)(k+2)(k+3) + (k+1)(k+2)(k+3)
\qquad \text{(by the inductive hypothesis)} \\
&=(k+1)(k+2)(k+3)\left(\frac{k}{4}+1\right) \\
&=(k+1)(k+2)(k+3)\left(\frac{k+4}{4}\right) \\
&=\frac{1}{4}(k+1)(k+2)(k+3)(k+4)
\end{align*}

This is the right-hand side with $n=k+1$, so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all positive integers $n$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q4]

Prove by induction that for $n \in \mathbb{Z}^{+}$
\[
\sum_{r=1}^{n} r\,3^{r-1} = \frac{1 + (2n - 1)3^n}{4}
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
Let
\[
S_n=\sum_{r=1}^{n} r\,3^{r-1}
\]

We prove by induction that
\[
S_n=\frac{1+(2n-1)3^n}{4}
\]

for all $n\in\mathbb{Z}^+$.

\textbf{Base case:} $n=1$. The left-hand side is
\[
\sum_{r=1}^{1} r\,3^{r-1}=1\times 3^0=1
\]

and the right-hand side is
\[
\frac{1+(2\times 1-1)3^1}{4}=\frac{1+3}{4}=1
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k\in\mathbb{Z}^+$. That is, assume
\[
\sum_{r=1}^{k} r\,3^{r-1}=\frac{1+(2k-1)3^k}{4}
\]

\textbf{Inductive step:} Starting with the left-hand side for $n=k+1$:
\begin{align*}
\sum_{r=1}^{k+1} r\,3^{r-1}
&= \sum_{r=1}^{k} r\,3^{r-1} + (k+1)3^k \\
&= \frac{1+(2k-1)3^k}{4} + (k+1)3^k
\end{align*}

using the inductive hypothesis.

Now write the second term over a denominator of $4$:
\begin{align*}
\sum_{r=1}^{k+1} r\,3^{r-1}
&= \frac{1+(2k-1)3^k}{4} + \frac{4(k+1)3^k}{4} \\
&= \frac{1+\bigl((2k-1)+4k+4\bigr)3^k}{4} \\
&= \frac{1+(6k+3)3^k}{4} \\
&= \frac{1+3(2k+1)3^k}{4} \\
&= \frac{1+(2k+1)3^{k+1}}{4}
\end{align*}

This is the right-hand side with $n=k+1$, since
\[
\frac{1+(2(k+1)-1)3^{k+1}}{4}=\frac{1+(2k+1)3^{k+1}}{4}
\]

so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all $n \in \mathbb{Z}^{+}$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q5]

\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Prove by induction that, for all integers $n \ge 1$
  \[
  \sum_{r=1}^{n} r^3 = \frac{1}{4}n^2(n+1)^2
  \]
  \marks[-20pt]{6}
  \item Hence, or otherwise, write down a factorised expression for the sum of the first $2n$ cubes
  \[
  1^3 + 2^3 + 3^3 + \cdots + (2n)^3
  \]
  \marks[-20pt]{1}
  \item Use the formula in part (a) to write down a factorised expression for the sum of the first $n$ even cubes
  \[
  2^3 + 4^3 + 6^3 + \cdots + (2n)^3
  \]
  \marks[-20pt]{2}
  \item Hence, or otherwise, show that
  \[
  1^3 + 3^3 + 5^3 + \cdots + (2n-1)^3 = an^2(bn^2-1)
  \]
  where $a$ and $b$ are rational numbers to be determined. \marks{3} \\
\end{enumerate}

\vspace*{10pt}

\begin{djsSolution}
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item We prove the result by mathematical induction.

  \textbf{Base case:} $n=1$. The left-hand side is
  \[
  \sum_{r=1}^{1} r^3 = 1^3 = 1
  \]

  and the right-hand side is
  \[
  \frac{1}{4}(1)^2(1+1)^2=\frac14\times 1\times 4=1
  \]

  so the statement is true for $n=1$.

  \textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k \ge 1$. That is, assume
  \[
  \sum_{r=1}^{k} r^3=\frac14 k^2(k+1)^2
  \]

  \textbf{Inductive step:} Consider $n=k+1$:
  \begin{align*}
  \sum_{r=1}^{k+1} r^3
  &= \sum_{r=1}^{k} r^3 + (k+1)^3 \\
  &= \frac14 k^2(k+1)^2 + (k+1)^3 \\
  &= (k+1)^2\left(\frac{k^2}{4}+k+1\right) \\
  &= (k+1)^2\left(\frac{k^2+4k+4}{4}\right) \\
  &= \frac14 (k+1)^2(k+2)^2
  \end{align*}

  This is the right-hand side with $n=k+1$, so the statement is true for $n=k+1$.

  \textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all integers $n \ge 1$.

  \item Using part (a) with $n$ replaced by $2n$:
  \begin{align*}
  1^3+2^3+3^3+\cdots +(2n)^3
  &= \sum_{r=1}^{2n} r^3 \\
  &= \frac14 (2n)^2(2n+1)^2 \\
  &= n^2(2n+1)^2
  \end{align*}

  \item The first $n$ even cubes are
  \[
  2^3+4^3+6^3+\cdots +(2n)^3 = \sum_{r=1}^{n} (2r)^3
  \]

  Hence
  \begin{align*}
  \sum_{r=1}^{n} (2r)^3
  &= 8\sum_{r=1}^{n} r^3 \\
  &= 8\left(\frac14 n^2(n+1)^2\right) \\
  &= 2n^2(n+1)^2
  \end{align*}

  \item The odd cubes are found by subtracting the even cubes from the first $2n$ cubes:
  \begin{align*}
  1^3+3^3+5^3+\cdots +(2n-1)^3
  &= \left(1^3+2^3+\cdots +(2n)^3\right) - \left(2^3+4^3+\cdots +(2n)^3\right) \\
  &= n^2(2n+1)^2 - 2n^2(n+1)^2 \\
  &= n^2\left((2n+1)^2-2(n+1)^2\right) \\
  &= n^2\left((4n^2+4n+1)-(2n^2+4n+2)\right) \\
  &= n^2(2n^2-1)
  \end{align*}

  so the sum is in the form $an^2(bn^2-1)$ with $a=1$, $b=2$.
\end{enumerate}
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q6]

\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Prove by induction that, for all positive integers $n$
  \[
  \sum_{r=1}^{n} \frac{1}{(2r-1)(2r+1)}=\frac{n}{2n+1}
  \]
  \marks[-30pt]{6}
  \item Hence, show that, for all positive integers $n$
  \[
  \sum_{r=n+1}^{2n} \frac{1}{(2r-1)(2r+1)}=\frac{n}{(an+b)(cn+d)}
  \]
  where $a$, $b$, $c$ and $d$ are integers to be determined. \marks{2} \\
\end{enumerate}

\vspace*{10pt}

\begin{djsSolution}
\begin{enumerate}[label=\textbf{(\alph*)}]
\item Let
\[
S_n=\sum_{r=1}^{n}\frac{1}{(2r-1)(2r+1)}
\]

We prove by induction that
\[
S_n=\frac{n}{2n+1}
\]

for all positive integers $n$.

\textbf{Base case:} $n=1$. The left-hand side is
\[
S_1=\frac{1}{(2\times1-1)(2\times1+1)}=\frac{1}{1\times3}=\frac13
\]

and the right-hand side is
\[
\frac{1}{2\times1+1}=\frac13
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k$ is a positive integer. That is, assume
\[
S_k=\sum_{r=1}^{k}\frac{1}{(2r-1)(2r+1)}=\frac{k}{2k+1}
\]

\textbf{Inductive step:} Consider $S_{k+1}$:
\begin{align*}
S_{k+1}
&=\sum_{r=1}^{k+1}\frac{1}{(2r-1)(2r+1)} \\
&=\sum_{r=1}^{k}\frac{1}{(2r-1)(2r+1)}+\frac{1}{(2(k+1)-1)(2(k+1)+1)} \\
&=S_k+\frac{1}{(2k+1)(2k+3)}
\end{align*}

Using the inductive hypothesis:
\begin{align*}
S_{k+1}
&=\frac{k}{2k+1}+\frac{1}{(2k+1)(2k+3)} \\
&=\frac{k(2k+3)+1}{(2k+1)(2k+3)} \\
&=\frac{2k^2+3k+1}{(2k+1)(2k+3)} \\
&=\frac{(2k+1)(k+1)}{(2k+1)(2k+3)} \\
&=\frac{k+1}{2k+3}
\end{align*}

This is the right-hand side with $n=k+1$, since
\[
\frac{k+1}{2k+3}=\frac{k+1}{2(k+1)+1}
\]

so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all positive integers $n$.

\item Using part (a):
\[
\sum_{r=n+1}^{2n}\frac{1}{(2r-1)(2r+1)}
=
\sum_{r=1}^{2n}\frac{1}{(2r-1)(2r+1)}
-
\sum_{r=1}^{n}\frac{1}{(2r-1)(2r+1)}
\]

So
\begin{align*}
\sum_{r=n+1}^{2n}\frac{1}{(2r-1)(2r+1)}
&=\frac{2n}{2(2n)+1}-\frac{n}{2n+1} \\
&=\frac{2n}{4n+1}-\frac{n}{2n+1} \\
&=\frac{2n(2n+1)-n(4n+1)}{(4n+1)(2n+1)} \\
&=\frac{4n^2+2n-4n^2-n}{(4n+1)(2n+1)} \\
&=\frac{n}{(4n+1)(2n+1)}
\end{align*}

so the sum is in the form $\frac{n}{(an+b)(cn+d)}$ with $a=4$, $b=1$, $c=2$, $d=1$.

\end{enumerate}
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q7]

Prove by induction that, for $n \ge 1$
\[
\sum_{r=1}^{n} \frac{r}{2^r} = 2 - \frac{n+2}{2^n}
\]
\marks[-20pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We prove the result by induction.

\textbf{Base case:} $n=1$. The left-hand side is
\[
\sum_{r=1}^{1}\frac{r}{2^r}=\frac{1}{2}
\]

and the right-hand side is
\[
2-\frac{1+2}{2^1}=2-\frac{3}{2}=\frac{1}{2}
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k\ge 1$. That is, assume
\[
\sum_{r=1}^{k}\frac{r}{2^r}=2-\frac{k+2}{2^k}
\]

\textbf{Inductive step:} We must show that
\[
\sum_{r=1}^{k+1}\frac{r}{2^r}=2-\frac{(k+1)+2}{2^{k+1}}
\]

Starting from the left-hand side:
\begin{align*}
\sum_{r=1}^{k+1}\frac{r}{2^r}
&=\sum_{r=1}^{k}\frac{r}{2^r}+\frac{k+1}{2^{k+1}}\\
&= \left(2-\frac{k+2}{2^k}\right)+\frac{k+1}{2^{k+1}}
\qquad \text{(by the inductive hypothesis)}\\
&=2-\frac{2k+4}{2^{k+1}}+\frac{k+1}{2^{k+1}}\\
&=2-\frac{2k+4-(k+1)}{2^{k+1}}\\
&=2-\frac{k+3}{2^{k+1}}\\
&=2-\frac{(k+1)+2}{2^{k+1}}
\end{align*}

This is the right-hand side with $n=k+1$, so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all $n \ge 1$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q8]

Prove by induction that for $n \in \mathbb{Z}^{+}$
\[
\sum_{r=1}^{n} \frac{1}{r(r+1)(r+2)} = \frac{n(n+3)}{4(n+1)(n+2)}
\]
\marks[-30pt]{6}

\vspace*{10pt}

\begin{djsSolution}
We prove the result by induction on $n$.

\textbf{Base case:} $n=1$. The left-hand side is
\[
\sum_{r=1}^{1} \frac{1}{r(r+1)(r+2)}=\frac{1}{1\times 2\times 3}=\frac16
\]

and the right-hand side is
\[
\frac{1(1+3)}{4(1+1)(1+2)}
=\frac{4}{4\times 2\times 3}
=\frac16
\]

so the statement is true for $n=1$.

\textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k\in\mathbb{Z}^+$. That is, assume
\[
\sum_{r=1}^{k} \frac{1}{r(r+1)(r+2)}
=
\frac{k(k+3)}{4(k+1)(k+2)}
\]

\textbf{Inductive step:} Starting with the left-hand side for $n=k+1$:
\begin{align*}
\sum_{r=1}^{k+1} \frac{1}{r(r+1)(r+2)}
&=
\sum_{r=1}^{k} \frac{1}{r(r+1)(r+2)}
+\frac{1}{(k+1)(k+2)(k+3)} \\
&=
\frac{k(k+3)}{4(k+1)(k+2)}
+\frac{1}{(k+1)(k+2)(k+3)}
\qquad \text{(by the inductive hypothesis)} \\
&=
\frac{k(k+3)(k+3)}{4(k+1)(k+2)(k+3)}
+\frac{4}{4(k+1)(k+2)(k+3)} \\
&=
\frac{k(k+3)^2+4}{4(k+1)(k+2)(k+3)} \\
&=
\frac{k(k^2+6k+9)+4}{4(k+1)(k+2)(k+3)} \\
&=
\frac{k^3+6k^2+9k+4}{4(k+1)(k+2)(k+3)} \\
&=
\frac{(k+1)^2(k+4)}{4(k+1)(k+2)(k+3)} \\
&=
\frac{(k+1)(k+4)}{4(k+2)(k+3)}
\end{align*}

This is the right-hand side with $n=k+1$, since
\[
\frac{(k+1)(k+4)}{4(k+2)(k+3)}
=
\frac{(k+1)\big((k+1)+3\big)}{4\big((k+1)+1\big)\big((k+1)+2\big)}
\]

so the statement is true for $n=k+1$.

\textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all $n \in \mathbb{Z}^{+}$.
\end{djsSolution}

\questionitem
% [Source: _QBT___Solns__Proof_by_Induction___Series, Q9]

\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Prove by induction that for all positive integers $n$
  \[
  \sum_{r=1}^{n} (2r-1)^3 = n^2(2n^2-1)
  \]
  \marks[-20pt]{6}
  \item Given that
  \[
  \sum_{r=1}^{2n} (2r-1)^3 = k\sum_{r=1}^{n} (2r-1)^3
  \]
  show that
  \[
  n^2 = \frac{k-4}{2(k-16)}
  \]
  \marks[-20pt]{5}
\end{enumerate}

\vspace*{10pt}

\begin{djsSolution}
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Let
  \[
  S_n=\sum_{r=1}^{n}(2r-1)^3
  \]

  We prove by induction that
  \[
  S_n=n^2(2n^2-1)
  \]

  for all positive integers $n$.

  \textbf{Base case:} $n=1$. The left-hand side is
  \[
  S_1=(2\times1-1)^3=1^3=1
  \]

  and the right-hand side is
  \[
  1^2(2\times1^2-1)=1(2-1)=1
  \]

  so the statement is true for $n=1$.

  \textbf{Inductive hypothesis:} Assume the statement is true for $n=k$, where $k$ is a positive integer. That is, assume
  \[
  S_k=\sum_{r=1}^{k}(2r-1)^3=k^2(2k^2-1)
  \]

  \textbf{Inductive step:} Starting from $S_{k+1}$:
  \begin{align*}
  S_{k+1}
  &=\sum_{r=1}^{k+1}(2r-1)^3 \\
  &=\sum_{r=1}^{k}(2r-1)^3+(2(k+1)-1)^3 \\
  &=S_k+(2k+1)^3
  \end{align*}

  Now use the inductive hypothesis:
  \begin{align*}
  S_{k+1}
  &=k^2(2k^2-1)+(2k+1)^3 \\
  &=2k^4-k^2+(8k^3+12k^2+6k+1) \\
  &=2k^4+8k^3+11k^2+6k+1
  \end{align*}

  We now factor this in the required form:
  \begin{align*}
  (k+1)^2\bigl(2(k+1)^2-1\bigr)
  &=(k+1)^2(2k^2+4k+1) \\
  &=(k^2+2k+1)(2k^2+4k+1) \\
  &=2k^4+8k^3+11k^2+6k+1
  \end{align*}

  Hence
  \[
  S_{k+1}=(k+1)^2\bigl(2(k+1)^2-1\bigr)
  \]

  so the statement is true for $n=k+1$.

  \textbf{Conclusion:} If the statement is true for $n=k$, then it is true for $n=k+1$. Since it is true for $n=1$, by the principle of mathematical induction the statement is true for all positive integers $n$.

  \item From part (a):
  \[
  \sum_{r=1}^{n}(2r-1)^3=n^2(2n^2-1)
  \]

  With $n$ replaced by $2n$:
  \begin{align*}
  \sum_{r=1}^{2n}(2r-1)^3
  &=(2n)^2\bigl(2(2n)^2-1\bigr) \\
  &=4n^2(8n^2-1)
  \end{align*}

  We are given that
  \[
  \sum_{r=1}^{2n}(2r-1)^3=k\sum_{r=1}^{n}(2r-1)^3
  \]

  Substitute the formula from part (a):
  \[
  4n^2(8n^2-1)=k\bigl(n^2(2n^2-1)\bigr)
  \]

  Since $n$ is a positive integer, $n^2\neq 0$, so divide through by $n^2$:
  \[
  4(8n^2-1)=k(2n^2-1)
  \]

  Expand:
  \[
  32n^2-4=2kn^2-k
  \]

  Rearranging:
  \begin{align*}
  32n^2-2kn^2&=4-k \\
  (32-2k)n^2&=4-k
  \end{align*}

  Multiply both sides by $-1$:
  \[
  (2k-32)n^2=k-4
  \]

  Hence
  \[
  n^2=\frac{k-4}{2k-32}=\frac{k-4}{2(k-16)}
  \]

\end{enumerate}
\end{djsSolution}

% [Source: _QBT___Solns__Proof_by_Induction___Series, Q10]

Prove by induction that for all positive integers $n$
\[
\sum_{r=1}^{n} \log\left(1-\frac{1}{4r^2}\right)=\log\left(\frac{(2n)!(2n+1)!}{2^{4n}(n!)^4}\right)
\]
\marks[-30pt]{6}

\vspace*{10pt}

\begin{djsSolution}
Let
\[
S_n=\sum_{r=1}^{n}\log\left(1-\frac{1}{4r^2}\right)
\]

We prove by induction that
\[
S_n=\log\left(\frac{(2n)!(2n+1)!}{2^{4n}(n!)^4}\right)
\]

for all positive integers \(n\).

\textbf{Base case:} \(n=1\). The left-hand side is
\[
S_1=\log\left(1-\frac14\right)=\log\left(\frac34\right)
\]

and the right-hand side is
\[
\log\left(\frac{(2\times1)!(2\times1+1)!}{2^{4\times1}(1!)^4}\right)
=\log\left(\frac{2!\,3!}{2^4}\right)
=\log\left(\frac{2\times6}{16}\right)
=\log\left(\frac34\right)
\]

so the statement is true for \(n=1\).

\textbf{Inductive hypothesis:} Assume the statement is true for \(n=k\), where \(k\) is a positive integer. That is, assume
\[
S_k=\log\left(\frac{(2k)!(2k+1)!}{2^{4k}(k!)^4}\right)
\]

\textbf{Inductive step:} We must show that
\[
S_{k+1}=\log\left(\frac{(2k+2)!(2k+3)!}{2^{4k+4}((k+1)!)^4}\right)
\]

Starting from the definition of \(S_{k+1}\):
\[
S_{k+1}=S_k+\log\left(1-\frac{1}{4(k+1)^2}\right)
\]

First simplify the extra term:
\begin{align*}
1-\frac{1}{4(k+1)^2}
&=\frac{4(k+1)^2-1}{4(k+1)^2} \\
&=\frac{(2k+2)^2-1}{(2k+2)^2} \\
&=\frac{(2k+1)(2k+3)}{(2k+2)^2}
\end{align*}

So, using the inductive hypothesis and the law \(\log a+\log b=\log(ab)\):
\begin{align*}
S_{k+1}
&=\log\left(\frac{(2k)!(2k+1)!}{2^{4k}(k!)^4}\right)
+\log\left(\frac{(2k+1)(2k+3)}{(2k+2)^2}\right) \\
&=\log\left(\frac{(2k)!(2k+1)!(2k+1)(2k+3)}{2^{4k}(k!)^4(2k+2)^2}\right)
\end{align*}

Now use the factorial identities:
\[
(2k+2)!=(2k+2)(2k+1)(2k)!
\]

and
\[
(2k+3)!=(2k+3)(2k+2)(2k+1)!
\]

Multiplying these together gives
\begin{align*}
(2k+2)!(2k+3)!
&=(2k+2)(2k+1)(2k)!\,(2k+3)(2k+2)(2k+1)! \\
&=(2k+2)^2(2k+1)(2k+3)(2k)!(2k+1)!
\end{align*}

Therefore
\[
(2k)!(2k+1)!(2k+1)(2k+3)=\frac{(2k+2)!(2k+3)!}{(2k+2)^2}
\]

Substituting this into the expression for \(S_{k+1}\):
\[
S_{k+1}
=\log\left(\frac{(2k+2)!(2k+3)!}{2^{4k}(k!)^4(2k+2)^4}\right)
\]

Now simplify the denominator:
\[
(2k+2)^4=[2(k+1)]^4=2^4(k+1)^4
\]

and
\[
((k+1)!)^4=((k+1)k!)^4=(k+1)^4(k!)^4
\]

Hence
\[
2^{4k}(k!)^4(2k+2)^4
=2^{4k}\times 2^4\times (k+1)^4(k!)^4
=2^{4k+4}((k+1)!)^4
\]

So
\[
S_{k+1}
=\log\left(\frac{(2k+2)!(2k+3)!}{2^{4k+4}((k+1)!)^4}\right)
\]

This is the right-hand side with \(n=k+1\), so the statement is true for \(n=k+1\).

\textbf{Conclusion:} If the statement is true for \(n=k\), then it is true for \(n=k+1\). Since it is true for \(n=1\), by the principle of mathematical induction the statement is true for all positive integers \(n\).
\end{djsSolution}
